```ts
/**
 * Hash puro: mesmos números de entrada sempre produzem a mesma saída, sem
 * estado compartilhado entre chamadas — ao contrário de um gerador sequencial,
 * não se desalinha se o React invocar o render mais de uma vez.

 * Vive aqui, e não no componente, porque a folha em SVG e a geometria em 3D
 * precisam ser a mesma planta. Mudar esta função muda as duas.
 */
export function pseudoRandom(...parts: number[]) {
  let h = 0x9e3779b9;
  for (const part of parts) {
    h = Math.imul(h ^ part, 2654435761);
    h = (h << 13) | (h >>> 19);
  }
  h = Math.imul(h ^ (h >>> 16), 2246822519);
  h ^= h >>> 13;
  return (h >>> 0) / 0xffffffff;
}
```

```ts
// app/_3d/registers.ts
/**
 * Os dois modos de ver a mesma geometria: prancha (traço de nanquim sobre
 * papel) e lâmina (campo escuro de microscopia). O registro é função da
 * posição do scroll, não do ato — a câmera atravessa seção papel com a
 * geometria do ato em quadro, e ali ela precisa virar traço.
 *
 * Blending não entra aqui de propósito: é propriedade de material e não
 * interpola, então trocá-lo no meio da travessia daria um pop. Os materiais
 * têm blending fixo e o que interpola é cor e alpha. `haloAlpha` chega a zero
 * no papel exatamente para que o blending aditivo do halo deixe de importar lá.
 */
export type Register = "prancha" | "lamina";

export type RegisterUniforms = {
  stroke: [number, number, number];
  strokeAlpha: number;
  halo: [number, number, number];
  haloAlpha: number;
};

/** --ink #10251b */
const INK: [number, number, number] = [0x10 / 255, 0x25 / 255, 0x1b / 255];
/** --acid #d9ff63 */
const ACID: [number, number, number] = [0xd9 / 255, 0xff / 255, 0x63 / 255];
/** --coral #ff7657 */
const CORAL: [number, number, number] = [0xff / 255, 0x76 / 255, 0x57 / 255];

const clamp01 = (v: number) => (v < 0 ? 0 : v > 1 ? 1 : v);

const lerp = (a: number, b: number, t: number) => a + (b - a) * t;

const lerp3 = (
  a: [number, number, number],
  b: [number, number, number],
  t: number,
): [number, number, number] => [
  lerp(a[0], b[0], t),
  lerp(a[1], b[1], t),
  lerp(a[2], b[2], t),
];

export function mixRegister(t: number): RegisterUniforms {
  const k = clamp01(t);
  return {
    stroke: lerp3(INK, ACID, k),
    strokeAlpha: lerp(0.14, 0.5, k),
    halo: lerp3(CORAL, CORAL, k),
    haloAlpha: lerp(0, 0.55, k),
  };
}
```

```ts
// app/_3d/progress.ts
import type { Register } from "./registers";

export type ActId = "branch" | "key" | "trichome" | "territory" | "clade";

/** Não se chama `Window`: sombrearia o global do DOM, que o diretor usa. */
export type ActWindow = { id: ActId; top: number; height: number };

export type SectionWindow = { top: number; height: number; register: Register };

const clamp01 = (v: number) => (v < 0 ? 0 : v > 1 ? 1 : v);

/**
 * O ato ativo é o que contém o centro da viewport. Fora de qualquer janela a
 * resposta é `null`: é travessia, e o diretor mantém em quadro a geometria do
 * ato anterior, retintada. Devolver o ato mais próximo daria um salto de
 * câmera no meio da seção papel.
 */
export function resolveAct(
  windows: ActWindow[],
  center: number,
): { id: ActId; progress: number } | null {
  for (const w of windows) {
    if (center < w.top || center > w.top + w.height) continue;
    return { id: w.id, progress: w.height > 0 ? clamp01((center - w.top) / w.height) : 0 };
  }
  return null;
}

/**
 * Quanto de lâmina, de 0 a 1. `band` é a largura da zona de mistura em volta
 * da fronteira entre seções, em pixels de documento: sem ela o registro
 * trocaria de golpe e o traço piscaria de tinta para ácido.
 */
export function resolveRegister(
  sections: SectionWindow[],
  center: number,
  band: number,
): number {
  if (sections.length === 0) return 0;

  const alvo = (y: number) => {
    for (const s of sections) {
      if (y < s.top || y > s.top + s.height) continue;
      return s.register === "lamina" ? 1 : 0;
    }
    // Acima da primeira ou abaixo da última seção: usa a mais próxima.
    const first = sections[0];
    const last = sections[sections.length - 1];
    const near = y < first.top ? first : last;
    return near.register === "lamina" ? 1 : 0;
  };

  if (band <= 0) return alvo(center);

  // Média de três amostras dentro da banda: dá uma rampa contínua na fronteira
  // sem precisar saber de que lado dela estamos.
  const meia = band / 2;
  return clamp01((alvo(center - meia) + alvo(center) + alvo(center + meia)) / 3);
}
```

```ts
// app/_3d/spring.ts
/**
 * Mola criticamente amortecida, integração semi-implícita.
 *
 * A câmera não segue o scroll direto porque jitter de trackpad sacudiria o
 * quadro, e não usa mola subamortecida porque overshoot em travelling de
 * câmera dá náusea. Amortecimento crítico é o único regime que converge rápido
 * e não passa do ponto.
 *
 * `dt` é clampado: uma aba que volta do background entrega um delta enorme, e
 * integração explícita com dt grande divergiria.
 */
const DT_MAX = 1 / 20;

export function stepSpring(
  current: number,
  velocity: number,
  target: number,
  omega: number,
  dt: number,
): { value: number; velocity: number } {
  const h = Math.min(dt, DT_MAX);
  // Semi-implícito: v primeiro, x com o v já atualizado. Estável para h*omega
  // moderado, ao contrário do Euler explícito.
  const denom = 1 + 2 * omega * h + omega * omega * h * h;
  const nextVelocity = (velocity + omega * omega * h * (target - current)) / denom;
  const nextValue = current + h * nextVelocity;
  return { value: nextValue, velocity: nextVelocity };
}
```

```ts
// app/_3d/glow.ts
import { DataTexture, LinearFilter, RGBAFormat, type Texture } from "three";

/**
 * O brilho do registro lâmina não vem de post-processing — vem deste sprite.
 * É um passe a menos, um framebuffer a menos e dezenas de KB a menos que
 * `UnrealBloomPass`, e a diferença visual num halo pequeno é desprezível.
 *
 * A rampa é separada do canvas de propósito: assim o núcleo é puro e dá para
 * testá-lo sem DOM.
 */
export function haloPixels(size: number): Uint8ClampedArray {
  const px = new Uint8ClampedArray(size * size * 4);
  const c = size / 2;
  for (let y = 0; y < size; y += 1) {
    for (let x = 0; x < size; x += 1) {
      const dx = x + 0.5 - c;
      const dy = y + 0.5 - c;
      const d = Math.sqrt(dx * dx + dy * dy) / c;
      // Quártica: cai mais rápido que linear perto da borda, então o halo tem
      // núcleo denso e franja longa — o que faz leitura de brilho.
      const fall = d >= 1 ? 0 : (1 - d) * (1 - d) * (1 - d) * (1 - d);
      const i = (y * size + x) * 4;
      px[i] = 255;
      px[i + 1] = 255;
      px[i + 2] = 255;
      px[i + 3] = Math.round(fall * 255);
    }
  }
  return px;
}

export function haloTexture(size = 64): Texture {
  const texture = new DataTexture(haloPixels(size), size, size, RGBAFormat);
  texture.minFilter = LinearFilter;
  texture.magFilter = LinearFilter;
  texture.needsUpdate = true;
  return texture;
}
```

```ts
// app/_3d/director.ts
import { PerspectiveCamera, Scene, Vector3, WebGLRenderer } from "three";
import { mixRegister } from "./registers";
import { resolveAct, resolveRegister, type ActWindow, type SectionWindow } from "./progress";
import { stepSpring } from "./spring";
import type { Act, ActContext } from "./types";

export type DirectorOptions = {
  /** Chamado quando o ato ativo muda; alimenta a barra de escala. */
  onAct: (id: string | null) => void;
  reducedMotion: boolean;
  compact: boolean;
};

const OMEGA = 6;
const BAND = 220;

export function createDirector(canvas: HTMLCanvasElement, opts: DirectorOptions) {
  const renderer = new WebGLRenderer({ canvas, alpha: true, antialias: !opts.compact });
  renderer.setClearAlpha(0);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, opts.compact ? 1 : 1.75));

  const scene = new Scene();
  const camera = new PerspectiveCamera(38, 1, 0.1, 100);

  const acts: Act[] = [];
  const ctx: ActContext = {
    budget: { particles: opts.compact ? 2000 : 8000, segments: opts.compact ? 32 : 96 },
    pointer: () => pointer,
  };

  let pointer: { x: number; y: number } | null = null;
  let frame = 0;
  let disposed = false;
  let last = performance.now();
  const start = last;

  /** Estado da mola, por eixo, para posição e alvo da câmera. */
  const cam = {
    pos: [0, 0, 3].map((v) => ({ value: v, velocity: 0 })),
    tgt: [0, 0, 0].map((v) => ({ value: v, velocity: 0 })),
  };

  let activeId: string | null = null;
  let windows: ActWindow[] = [];
  let sections: SectionWindow[] = [];

  /** Geometria fica em memória; o layout é que é remedido. Um `resize` não
      reconstrói ato nenhum. */
  const measure = () => {
    const doc = document.documentElement;
    renderer.setSize(window.innerWidth, window.innerHeight, false);
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();

    windows = acts.flatMap((act) => {
      const el = document.querySelector(act.anchor);
      if (!el) return [];
      const rect = el.getBoundingClientRect();
      return [{ id: act.id, top: rect.top + window.scrollY, height: rect.height }];
    });

    sections = Array.from(document.querySelectorAll<HTMLElement>("[data-register]")).map(
      (el) => {
        const rect = el.getBoundingClientRect();
        return {
          top: rect.top + window.scrollY,
          height: rect.height,
          register: el.dataset.register === "lamina" ? "lamina" : "prancha",
        };
      },
    );

    void doc;
  };

  const draw = (now: number) => {
    frame = 0;
    if (disposed) return;

    const dt = Math.max((now - last) / 1000, 0);
    last = now;
    const t = (now - start) / 1000;

    const center = window.scrollY + window.innerHeight / 2;
    const active = resolveAct(windows, center);
    const registerMix = resolveRegister(sections, center, BAND);
    const uniforms = mixRegister(registerMix);

    if ((active?.id ?? null) !== activeId) {
      activeId = active?.id ?? null;
      opts.onAct(activeId);
    }

    // Na travessia o ato anterior continua em quadro, retintado — é o que faz
    // "a mesma estrutura, dois modos de ver" acontecer de fato.
    const inFrame = acts.find((a) => a.id === (activeId ?? acts[0]?.id));
    if (inFrame) {
      const progress = active?.progress ?? 1;
      inFrame.update(t, progress, uniforms);
      const station = inFrame.station(progress);
      const alvo = [station.position, station.target] as const;
      const springs = [cam.pos, cam.tgt] as const;
      for (let s = 0; s < 2; s += 1) {
        const v = alvo[s];
        const eixos = [v.x, v.y, v.z];
        for (let i = 0; i < 3; i += 1) {
          springs[s][i] = opts.reducedMotion
            ? { value: eixos[i], velocity: 0 }
            : stepSpring(springs[s][i].value, springs[s][i].velocity, eixos[i], OMEGA, dt);
        }
      }
      camera.position.set(cam.pos[0].value, cam.pos[1].value, cam.pos[2].value);
      camera.lookAt(new Vector3(cam.tgt[0].value, cam.tgt[1].value, cam.tgt[2].value));

      const clip = inFrame.frame?.();
      if (clip) {
        const r = clip.getBoundingClientRect();
        const y = window.innerHeight - r.bottom;
        renderer.setScissorTest(true);
        renderer.setScissor(r.left, y, r.width, r.height);
        renderer.setViewport(r.left, y, r.width, r.height);
      } else {
        renderer.setScissorTest(false);
        renderer.setViewport(0, 0, window.innerWidth, window.innerHeight);
      }
    }

    renderer.render(scene, camera);
    if (!opts.reducedMotion) schedule();
  };

  const schedule = () => {
    if (frame || disposed) return;
    frame = requestAnimationFrame(draw);
  };

  const onScroll = () => schedule();
  const onResize = () => {
    measure();
    schedule();
  };
  const onPointer = (event: PointerEvent) => {
    if (event.pointerType !== "mouse") return;
    pointer = {
      x: (event.clientX / window.innerWidth) * 2 - 1,
      y: -((event.clientY / window.innerHeight) * 2 - 1),
    };
  };
  const onVisibility = () => {
    if (document.hidden) {
      if (frame) cancelAnimationFrame(frame);
      frame = 0;
    } else {
      last = performance.now();
      schedule();
    }
  };
  const onLost = (event: Event) => {
    event.preventDefault();
    canvas.style.display = "none";
    disposed = true;
  };

  measure();
  schedule();
  window.addEventListener("scroll", onScroll, { passive: true });
  window.addEventListener("resize", onResize);
  window.addEventListener("pointermove", onPointer, { passive: true });
  document.addEventListener("visibilitychange", onVisibility);
  canvas.addEventListener("webglcontextlost", onLost);

  return {
    destroy() {
      disposed = true;
      if (frame) cancelAnimationFrame(frame);
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("resize", onResize);
      window.removeEventListener("pointermove", onPointer);
      document.removeEventListener("visibilitychange", onVisibility);
      canvas.removeEventListener("webglcontextlost", onLost);
      for (const act of acts) act.dispose();
      renderer.dispose();
    },
  };
}
```

```tsx
// app/_components/Stage.tsx
"use client";

import { useEffect, useRef, useState } from "react";
import type { Lang } from "@/data/site";

/**
 * O shell da cena. Renderiza canvas vazio e barra de escala; o diretor e o
 * three.js entram por `import()` depois do first paint, então o LCP da página
 * não muda. Se WebGL2 não existir, o canvas sai do DOM e a página fica
 * exatamente como era.
 */
export default function Stage({ lang }: { lang: Lang }) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [act, setAct] = useState<string | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    // saveData: quem pediu para economizar dados não pediu por atmosfera.
    const connection = (navigator as { connection?: { saveData?: boolean } }).connection;
    if (connection?.saveData) {
      canvas.remove();
      return;
    }

    if (!("WebGL2RenderingContext" in window)) {
      canvas.remove();
      return;
    }

    const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const compact = window.matchMedia("(max-width: 820px)").matches;

    let director: { destroy(): void } | null = null;
    let cancelled = false;

    const boot = async () => {
      try {
        const { createDirector } = await import("../_3d/director");
        if (cancelled) return;
        director = createDirector(canvas, { onAct: setAct, reducedMotion, compact });
      } catch {
        // Contexto que não sobe não é erro do usuário: a página inteira
        // funciona sem a cena, então o silêncio aqui é deliberado.
        canvas.remove();
      }
    };

    const idle = window.requestIdleCallback ?? ((cb: () => void) => setTimeout(cb, 200));
    const handle = idle(boot);

    return () => {
      cancelled = true;
      if (window.cancelIdleCallback && typeof handle === "number") {
        window.cancelIdleCallback(handle);
      }
      director?.destroy();
    };
  }, []);

  void lang;

  return (
    <>
      <canvas className="stage" ref={canvasRef} aria-hidden="true" />
      <div className="stage-scale" data-act={act ?? ""} />
    </>
  );
}
```

```js
// scripts/check-bundle.mjs
/**
 * O teto de 120 KB gzip para a cena é decisão de produto, não estética: esta é
 * uma landing de recolocação, e três dependências foram mantidas por anos de
 * propósito. Este script transforma o teto em número, para não virar promessa.
 */
import { gzipSync } from "node:zlib";
import { readdir, readFile } from "node:fs/promises";

const TETO = 120 * 1024;
const dir = "dist/_next/static/chunks";

const arquivos = await readdir(dir);
const medidos = await Promise.all(
  arquivos
    .filter((nome) => nome.endsWith(".js"))
    .map(async (nome) => {
      const bruto = await readFile(`${dir}/${nome}`);
      // O chunk da cena é o que contém three.js. Achar por conteúdo é mais
      // robusto que por nome: o hash muda a cada build.
      const three = bruto.includes("WebGLRenderer") || bruto.includes("THREE.WebGL");
      return { nome, gzip: gzipSync(bruto).length, three };
    }),
);

const cena = medidos.filter((m) => m.three);
if (cena.length === 0) {
  console.error("Nenhum chunk com three.js encontrado — o dynamic import quebrou?");
  process.exit(1);
}

const total = cena.reduce((soma, m) => soma + m.gzip, 0);
for (const m of cena) {
  console.log(`  ${m.nome}  ${(m.gzip / 1024).toFixed(1)} KB gzip`);
}
console.log(`\ncena: ${(total / 1024).toFixed(1)} KB gzip · teto ${(TETO / 1024).toFixed(0)} KB`);

if (total > TETO) {
  console.error(`\nESTOUROU em ${((total - TETO) / 1024).toFixed(1)} KB.`);
  process.exit(1);
}
```

```ts
// app/_3d/acts/branch.ts
import { pseudoRandom } from "../random";

export const FLOATS_PER_LEAFLET = 7;

/** Ráquis, pinas e folíolos, na mesma proporção do Frond em SVG. */
const RAQUIS = 0.72;
const LEAFLET_STEP = 0.026;

/**
 * Uma entrada por folíolo: posição (3), eixo de dobra (3) e o ângulo de dobra
 * (1), que a tigmonastia escreve em runtime. Sai como Float32Array porque é
 * atributo de instância — construir objetos aqui só para desmontá-los depois
 * seria trabalho para o GC a cada resize.
 */
export function buildLeafletTransforms(leaves: number, pinnae: number): Float32Array {
  const out: number[] = [];

  for (let leaf = 0; leaf < leaves; leaf += 1) {
    const yaw = (pseudoRandom(leaf, 11) - 0.5) * 2.4;
    const tilt = (pseudoRandom(leaf, 23) - 0.5) * 0.9;
    const scale = 0.6 + pseudoRandom(leaf, 31) * 0.4;

    for (let p = 0; p < pinnae; p += 1) {
      const t = 0.1 + (p / Math.max(pinnae - 1, 1)) * 0.88;
      const along = t * RAQUIS * scale;
      // Curvatura da ráquis: a mesma parábola do SVG, agora em Y.
      const droop = -RAQUIS * 0.05 * (4 * t * (1 - t)) - RAQUIS * 0.02 * t * t;
      const length = (0.09 + 0.1 * Math.sin(Math.PI * t)) * scale;
      const count = Math.max(2, Math.round(length / LEAFLET_STEP));

      for (const side of [-1, 1]) {
        const spread = side * (0.9 + pseudoRandom(leaf, p, side) * 0.21);
        for (let f = 1; f <= count; f += 1) {
          const reach = (f / count) * length;
          const x = along * Math.cos(yaw) - reach * Math.sin(yaw) * Math.cos(spread);
          const z = along * Math.sin(yaw) + reach * Math.cos(yaw) * Math.cos(spread);
          const y = droop * scale + reach * Math.sin(spread) * Math.sin(tilt);
          out.push(
            clampUnit(x),
            clampUnit(y),
            clampUnit(z),
            // Eixo de dobra: o folíolo gira em volta da própria pina.
            Math.cos(yaw),
            0,
            Math.sin(yaw),
            0,
          );
        }
      }
    }
  }

  return new Float32Array(out);
}

const clampUnit = (v: number) => (v > 1 ? 1 : v < -1 ? -1 : v);
```

```ts
// app/_3d/acts/key.ts
/** A forma real de `key` em data/site.ts. `mark` não interessa ao layout. */
type Lead = { goTo?: string; track?: string };
type Couplet = { id: string; leads: readonly Lead[] };

export type KeyNode = { x: number; y: number; z: number; depth: number };

/**
 * Layout de árvore binária em caixa unitária: profundidade desce em Y, irmãos
 * se separam em X, e o Z recebe um pequeno desvio por nível para que a
 * topologia leia como volume e não como diagrama achatado.
 *
 * A travessia carrega um conjunto de visitados porque a estrutura vem de dados
 * escritos à mão: um `leads` que aponte para trás não deve virar loop infinito
 * no meio do rAF.
 */
export function layoutKeyTree(couplets: readonly Couplet[]): Map<string, KeyNode> {
  const byId = new Map(couplets.map((c) => [c.id, c]));
  const layout = new Map<string, KeyNode>();
  const root = couplets[0];
  if (!root) return layout;

  let maxDepth = 0;
  const fila: Array<{ id: string; depth: number; lo: number; hi: number }> = [
    { id: root.id, depth: 0, lo: -1, hi: 1 },
  ];
  const vistos = new Set<string>();

  while (fila.length > 0) {
    const { id, depth, lo, hi } = fila.shift()!;
    if (vistos.has(id)) continue;
    vistos.add(id);

    const x = (lo + hi) / 2;
    maxDepth = Math.max(maxDepth, depth);
    layout.set(id, { x, y: 0, z: ((depth % 2) - 0.5) * 0.3, depth });

    // Um lead aponta para outro couplet ou termina numa trilha; nos dois casos
    // o destino é um nó do layout.
    const leads = byId.get(id)?.leads ?? [];
    const passo = (hi - lo) / Math.max(leads.length, 1);
    leads.forEach((lead, i) => {
      const filho = lead.goTo ?? lead.track;
      if (!filho) return;
      fila.push({ id: filho, depth: depth + 1, lo: lo + i * passo, hi: lo + (i + 1) * passo });
    });
  }

  // Y só dá para normalizar depois de conhecer a profundidade máxima.
  for (const node of layout.values()) {
    node.y = maxDepth > 0 ? 1 - (node.depth / maxDepth) * 2 : 0;
  }

  return layout;
}
```

```ts
// app/_3d/acts/trichome.ts
/**
 * Sólido de revolução a partir de um perfil `[raio, altura]`. É a forma mais
 * barata de tricoma glandular: pedúnculo fino, cabeça multicelular no topo.
 *
 * O giro fecha pelo índice e não repetindo o primeiro vértice do anel — um anel
 * com costura duplicada gasta vértice e cria uma emenda visível na normal.
 */
export function revolveProfile(
  profile: Array<[number, number]>,
  segments: number,
): { positions: Float32Array; indices: Uint16Array } {
  const positions = new Float32Array(profile.length * segments * 3);
  const indices = new Uint16Array((profile.length - 1) * segments * 6);

  for (let p = 0; p < profile.length; p += 1) {
    const [radius, height] = profile[p];
    for (let s = 0; s < segments; s += 1) {
      const a = (s / segments) * Math.PI * 2;
      const i = (p * segments + s) * 3;
      positions[i] = Math.cos(a) * radius;
      positions[i + 1] = height;
      positions[i + 2] = Math.sin(a) * radius;
    }
  }

  let k = 0;
  for (let p = 0; p < profile.length - 1; p += 1) {
    for (let s = 0; s < segments; s += 1) {
      const proximo = (s + 1) % segments;
      const a = p * segments + s;
      const b = p * segments + proximo;
      const c = (p + 1) * segments + proximo;
      const d = (p + 1) * segments + s;
      indices[k] = a;
      indices[k + 1] = b;
      indices[k + 2] = c;
      indices[k + 3] = a;
      indices[k + 4] = c;
      indices[k + 5] = d;
      k += 6;
    }
  }

  return { positions, indices };
}
```

```ts
// app/_3d/acts/territory.ts
export const BRASIL_BOUNDS = { west: -74, east: -33, south: -34, north: 6 } as const;

/**
 * Equirretangular, normalizada para a caixa unitária. Não é projeção para
 * medir: é para a silhueta ser reconhecível ao lado do texto que fala dela.
 * Norte fica em -z porque é para lá que a câmera olha na órbita descendente.
 */
export function projectEquirectangular(lon: number, lat: number): { x: number; z: number } {
  const { west, east, south, north } = BRASIL_BOUNDS;
  const x = ((lon - west) / (east - west)) * 2 - 1;
  const z = -(((lat - south) / (north - south)) * 2 - 1);
  return { x, z };
}
```

simplesmente não desenha nada.

```ts
// app/_3d/acts/clade.ts
import {
  CLADOGRAM_SUB_LANE,
  cladogramLanes,
  cladogramNodes,
  cladogramYears,
} from "@/data/cladogram";

/**
 * Teto de opacidade do ato. Cladograma em 3D fica mais bonito e menos legível
 * que em 2D; aqui o 3D é atmosfera, e a leitura da topologia continua sendo o
 * Cladogram.tsx, que já é correto. Este número é o contrato disso.
 */
export const ATMOSPHERIC_ALPHA_CEILING = 0.3;

export type CladePoint = { x: number; y: number; z: number };

/**
 * Ano no eixo X, faixa no eixo Y, desvio por faixa no Z para dar volume.
 *
 * Não assume árvore: o nó de 2020 é reticulação, com dois ancestrais e um
 * descendente, e um layout de árvore o colocaria em dois lugares.
 *
 * A faixa `"sub"` não está em `cladogramLanes` — é a graduação em curso
 * pendurada na linhagem de dados — então o Y dela vem de `CLADOGRAM_SUB_LANE`,
 * e um lookup ingênuo de faixa deixaria o nó `dataScience` de fora.
 */
export function layoutClade(): Map<string, CladePoint> {
  const yPorFaixa = new Map<string, number>(
    cladogramLanes.map((lane) => [lane.id as string, lane.y]),
  );
  yPorFaixa.set("sub", CLADOGRAM_SUB_LANE);

  const ys = [...yPorFaixa.values()];
  const yMin = Math.min(...ys);
  const yMax = Math.max(...ys);
  const { start, end } = cladogramYears;

  const norm = (v: number, lo: number, hi: number) =>
    hi > lo ? ((v - lo) / (hi - lo)) * 2 - 1 : 0;

  const ordemDaFaixa = new Map<string, number>(
    [...yPorFaixa.keys()].map((id, i) => [id, i]),
  );

  const layout = new Map<string, CladePoint>();
  for (const [i, node] of cladogramNodes.entries()) {
    const faixaY = yPorFaixa.get(node.lane) ?? yMin;
    const faixa = ordemDaFaixa.get(node.lane) ?? 0;
    layout.set(node.id, {
      x: norm(node.year, start, end),
      // Y invertido: no SVG o eixo cresce para baixo, em 3D cresce para cima.
      y: -norm(faixaY, yMin, yMax),
      // O termo com `i` desempata nós de mesmo ano e mesma faixa, que sem ele
      // ficariam exatamente sobrepostos.
      z: ((faixa % 2) - 0.5) * 0.4 + (i % 3) * 0.012,
    });
  }

  return layout;
}
```
